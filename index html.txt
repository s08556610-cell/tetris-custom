  <script>
    const COLS = 10;
    const ROWS = 20;
    const BLOCK = 24;

    const COLORS = {
      I: "#00f0f0",
      O: "#f0f000",
      T: "#a000f0",
      S: "#00f000",
      Z: "#f00000",
      J: "#0000f0",
      L: "#f0a000",
    };

    const SHAPES = {
      I: [
        [[0, 1], [1, 1], [2, 1], [3, 1]],
        [[2, 0], [2, 1], [2, 2], [2, 3]],
        [[0, 2], [1, 2], [2, 2], [3, 2]],
        [[1, 0], [1, 1], [1, 2], [1, 3]],
      ],
      O: [
        [[1, 0], [2, 0], [1, 1], [2, 1]],
        [[1, 0], [2, 0], [1, 1], [2, 1]],
        [[1, 0], [2, 0], [1, 1], [2, 1]],
        [[1, 0], [2, 0], [1, 1], [2, 1]],
      ],
      T: [
        [[1, 0], [0, 1], [1, 1], [2, 1]],
        [[1, 0], [1, 1], [2, 1], [1, 2]],
        [[0, 1], [1, 1], [2, 1], [1, 2]],
        [[1, 0], [0, 1], [1, 1], [1, 2]],
      ],
      S: [
        [[1, 0], [2, 0], [0, 1], [1, 1]],
        [[1, 0], [1, 1], [2, 1], [2, 2]],
        [[1, 1], [2, 1], [0, 2], [1, 2]],
        [[0, 0], [0, 1], [1, 1], [1, 2]],
      ],
      Z: [
        [[0, 0], [1, 0], [1, 1], [2, 1]],
        [[2, 0], [1, 1], [2, 1], [1, 2]],
        [[0, 1], [1, 1], [1, 2], [2, 2]],
        [[1, 0], [0, 1], [1, 1], [0, 2]],
      ],
      J: [
        [[0, 0], [0, 1], [1, 1], [2, 1]],
        [[1, 0], [2, 0], [1, 1], [1, 2]],
        [[0, 1], [1, 1], [2, 1], [2, 2]],
        [[1, 0], [1, 1], [0, 2], [1, 2]],
      ],
      L: [
        [[2, 0], [0, 1], [1, 1], [2, 1]],
        [[1, 0], [1, 1], [1, 2], [2, 2]],
        [[0, 1], [1, 1], [2, 1], [0, 2]],
        [[0, 0], [1, 0], [1, 1], [1, 2]],
      ],
    };

    const canvas = document.getElementById("board");
    const ctx = canvas.getContext("2d");
    const statusEl = document.getElementById("status");
    const scoreEl = document.getElementById("score");
    const linesEl = document.getElementById("lines");

    let grid, current, score, lines, gameOver, dropTimer, lastTime;

    function createGrid() {
      return Array.from({ length: ROWS }, () => Array(COLS).fill(null));
    }

    function getCells(piece) {
      return SHAPES[piece.kind][piece.rot].map(([dx, dy]) => [piece.x + dx, piece.y + dy]);
    }

    function isValid(piece) {
      return getCells(piece).every(([x, y]) => {
        if (x < 0 || x >= COLS || y >= ROWS) return false;
        if (y >= 0 && grid[y][x] !== null) return false;
        return true;
      });
    }

    function spawnPiece() {
      const kinds = Object.keys(SHAPES);
      const kind = kinds[Math.floor(Math.random() * kinds.length)];
      const piece = { kind, x: Math.floor(COLS / 2) - 2, y: 0, rot: 0 };
      if (!isValid(piece)) {
        current = null;
        gameOver = true;
      } else {
        current = piece;
      }
    }

    function resetGame() {
      grid = createGrid();
      score = 0;
      lines = 0;
      gameOver = false;
      dropTimer = 0;
      lastTime = performance.now();
      spawnPiece();
      updateUI();
    }

    function move(dx, dy) {
      if (gameOver || !current) return false;
      const next = { ...current, x: current.x + dx, y: current.y + dy };
      if (isValid(next)) {
        current = next;
        return true;
      }
      return false;
    }

    function rotate(dir = 1) {
      if (gameOver || !current) return false;
      const next = { ...current, rot: (current.rot + dir + 4) % 4 };
      if (isValid(next)) {
        current = next;
        return true;
      }
      return false;
    }

    function lockAndClear() {
      if (!current) return;
      for (const [x, y] of getCells(current)) {
        if (y >= 0 && y < ROWS && x >= 0 && x < COLS) {
          grid[y][x] = current.kind;
        }
      }
      const remaining = grid.filter(row => row.some(cell => cell === null));
      const cleared = ROWS - remaining.length;
      if (cleared > 0) {
        const emptyRows = Array.from({ length: cleared }, () => Array(COLS).fill(null));
        grid = [...emptyRows, ...remaining];
        lines += cleared;
        score += cleared * 100;
      }
      spawnPiece();
      updateUI();
    }

    function stepDown() {
      if (gameOver) return;
      if (!move(0, 1)) {
        lockAndClear();
      }
    }

    function hardDrop() {
      if (gameOver) return;
      while (move(0, 1)) {}
      lockAndClear();
    }

    function drawCell(x, y, color) {
      ctx.fillStyle = color;
      ctx.fillRect(x * BLOCK + 1, y * BLOCK + 1, BLOCK - 2, BLOCK - 2);
      ctx.strokeStyle = "rgba(255,255,255,0.25)";
      ctx.strokeRect(x * BLOCK + 1.5, y * BLOCK + 1.5, BLOCK - 3, BLOCK - 3);
    }

    function draw() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      // グリッド線
      ctx.strokeStyle = "#1e293b";
      ctx.lineWidth = 1;
      for (let x = 0; x <= COLS; x++) {
        ctx.beginPath();
        ctx.moveTo(x * BLOCK, 0);
        ctx.lineTo(x * BLOCK, canvas.height);
        ctx.stroke();
      }
      for (let y = 0; y <= ROWS; y++) {
        ctx.beginPath();
        ctx.moveTo(0, y * BLOCK);
        ctx.lineTo(canvas.width, y * BLOCK);
        ctx.stroke();
      }

      // 固定ブロック
      for (let y = 0; y < ROWS; y++) {
        for (let x = 0; x < COLS; x++) {
          if (grid[y][x]) {
            drawCell(x, y, COLORS[grid[y][x]]);
          }
        }
      }

      // 操作中ミノ
      if (current && !gameOver) {
        for (const [x, y] of getCells(current)) {
          if (y >= 0) {
            drawCell(x, y, COLORS[current.kind]);
          }
        }
      }
    }

    function updateUI() {
      scoreEl.textContent = score;
      linesEl.textContent = lines;
      if (gameOver) {
        statusEl.textContent = "GAME OVER";
        statusEl.className = "status-over";
      } else {
        statusEl.textContent = "PLAYING";
        statusEl.className = "status-playing";
      }
    }

    function loop(now = performance.now()) {
      const dt = now - lastTime;
      lastTime = now;
      if (!gameOver) {
        dropTimer += dt;
        if (dropTimer >= 600) {
          dropTimer = 0;
          stepDown();
        }
      }
      draw();
      requestAnimationFrame(loop);
    }

    document.addEventListener("keydown", (e) => {
      if (e.key === "ArrowLeft") { e.preventDefault(); move(-1, 0); }
      else if (e.key === "ArrowRight") { e.preventDefault(); move(1, 0); }
      else if (e.key === "ArrowDown") { e.preventDefault(); stepDown(); }
      else if (e.key === "ArrowUp" || e.key.toLowerCase() === "x") { e.preventDefault(); rotate(1); }
      else if (e.code === "Space") { e.preventDefault(); hardDrop(); }
      else if (e.key.toLowerCase() === "r") { e.preventDefault(); resetGame(); }
    });

    document.getElementById("btn-left").addEventListener("click", () => move(-1, 0));
    document.getElementById("btn-right").addEventListener("click", () => move(1, 0));
    document.getElementById("btn-down").addEventListener("click", () => stepDown());
    document.getElementById("btn-rot").addEventListener("click", () => rotate(1));
    document.getElementById("btn-drop").addEventListener("click", () => hardDrop());
    document.getElementById("btn-reset").addEventListener("click", () => resetGame());

    resetGame();
    requestAnimationFrame(loop);
  </script>
